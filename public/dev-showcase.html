const pipeline = [
  "project.scan",
  "handover.load",
  "routes.connect",
  "mobile.preview",
  "admin.metrics",
  "funding.review"
];

for (const step of pipeline) {
  build.show(step);
  appFrame.capture(step);
  checklist.mark(step, "visible");
}
